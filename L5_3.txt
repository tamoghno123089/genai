import os
from gpt4all import GPT4All
import chromadb
from chromadb.config import Settings
from google.generativeai import GenerativeModel

# -----------------------------
# 1. Initialize ChromaDB client
# -----------------------------
chroma_client = chromadb.Client(Settings(
    chroma_db_impl="duckdb+parquet",
    persist_directory="./chroma_store"  # local persistence
))

# Create or get a collection
collection = chroma_client.get_or_create_collection(name="rag_collection")

# -----------------------------
# 2. Gemini Embedding Function
# -----------------------------
import google.generativeai as genai

# Set your Gemini API key
genai.configure(api_key=os.environ["GEMINI_API_KEY"])

embedding_model = genai.GenerativeModel("text-embedding-004")

def get_embedding(text: str):
    """Generate embeddings using Gemini."""
    response = embedding_model.embed_content(text)
    return response["embedding"]

# -----------------------------
# 3. Add documents to ChromaDB
# -----------------------------
documents = [
    "GPT4All is a local LLM that runs on your desktop.",
    "Gemini is a Google AI model that provides embeddings and generative capabilities.",
    "ChromaDB is a vector database used for storing and retrieving embeddings."
]

for i, doc in enumerate(documents):
    emb = get_embedding(doc)
    collection.add(
        ids=[f"doc_{i}"],
        documents=[doc],
        embeddings=[emb]
    )

# -----------------------------
# 4. Query pipeline (RAG)
# -----------------------------
def rag_query(query: str):
    # Step 1: Embed query with Gemini
    query_emb = get_embedding(query)

    # Step 2: Retrieve top documents from ChromaDB
    results = collection.query(
        query_embeddings=[query_emb],
        n_results=3
    )

    retrieved_docs = results["documents"][0]

    # Step 3: Construct context
    context = "\n".join(retrieved_docs)

    # Step 4: Use GPT4All for final answer
    gpt4all_model = GPT4All("gpt4all-falcon-q4_0.gguf")  # example model file
    with gpt4all_model.chat_session() as session:
        prompt = f"Answer the following question using the context:\n\nContext:\n{context}\n\nQuestion: {query}\nAnswer:"
        response = session.prompt(prompt)
    
    return response

# -----------------------------
# 5. Example usage
# -----------------------------
if __name__ == "__main__":
    user_query = "What is ChromaDB used for?"
    answer = rag_query(user_query)
    print("Answer:", answer)
