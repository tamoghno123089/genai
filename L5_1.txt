## 📥 ingest.py (Load & Index PDFs)
```python
import os
import PyPDF2
from langchain.vectorstores import FAISS
from langchain.embeddings import OllamaEmbeddings
from langchain.text_splitter import RecursiveCharacterTextSplitter

DATA_DIR = "data"
INDEX_DIR = "vectorstore"

def load_pdfs():
    texts = []
    for file in os.listdir(DATA_DIR):
        if file.endswith(".pdf"):
            with open(os.path.join(DATA_DIR, file), "rb") as f:
                reader = PyPDF2.PdfReader(f)
                for page in reader.pages:
                    texts.append(page.extract_text())
    return texts

def build_index():
    texts = load_pdfs()
    splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=100)
    docs = splitter.create_documents(texts)

    embeddings = OllamaEmbeddings(model="llama3.2")
    vectorstore = FAISS.from_documents(docs, embeddings)
    vectorstore.save_local(INDEX_DIR)
    print("✅ Index built and saved.")

if __name__ == "__main__":
    build_index()
```

## 🔍 query.py (RAG Pipeline)
```python
from langchain.vectorstores import FAISS
from langchain.embeddings import OllamaEmbeddings
from langchain.llms import Ollama

INDEX_DIR = "vectorstore"

def rag_query(question: str):
    embeddings = OllamaEmbeddings(model="llama3.2")
    vectorstore = FAISS.load_local(INDEX_DIR, embeddings)

    retriever = vectorstore.as_retriever(search_kwargs={"k": 3})
    docs = retriever.get_relevant_documents(question)

    context = "\n\n".join([d.page_content for d in docs])
    prompt = f"Answer the question using the context below:\n\n{context}\n\nQuestion: {question}"

    llm = Ollama(model="llama3.2")
    response = llm(prompt)
    return response

if __name__ == "__main__":
    print(rag_query("Summarize the loan eligibility criteria."))
```

## 🎨 chat_ui.py (Streamlit Chat Interface)
```python
import streamlit as st
from query import rag_query

st.set_page_config(page_title="RAG PDF Chat", layout="wide")

st.title("📚 RAG PDF Chat with Ollama (Llama 3.2)")
st.write("Ask questions based on your uploaded PDFs.")

# Chat history
if "messages" not in st.session_state:
    st.session_state["messages"] = []

# Display chat history
for msg in st.session_state["messages"]:
    if msg["role"] == "user":
        st.chat_message("user").markdown(msg["content"])
    else:
        st.chat_message("assistant").markdown(msg["content"])

# User input
if prompt := st.chat_input("Type your question..."):
    st.session_state["messages"].append({"role": "user", "content": prompt})
    st.chat_message("user").markdown(prompt)

    # Get RAG answer
    answer = rag_query(prompt)
    st.session_state["messages"].append({"role": "assistant", "content": answer})
    st.chat_message("assistant").markdown(answer)
```
